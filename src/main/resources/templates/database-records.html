Academic Relational Queries & SQL Constructs
1. Multiple Table INNER JOIN (Tickets + Vehicles + Payments)
SELECT 
    t.ticket_number, 
    v.vehicle_number, 
    v.owner_name, 
    v.vehicle_type, 
    t.allocated_slots,
    p.amount,
    p.payment_method
FROM parking_tickets t
INNER JOIN vehicles v ON t.vehicle_id = v.vehicle_id
LEFT JOIN payments p ON t.ticket_id = p.ticket_id;
2. Aggregate Functions & GROUP BY (Revenue Distribution)
SELECT 
    v.vehicle_type,
    COUNT(t.ticket_id) AS total_sessions,
    COALESCE(SUM(p.amount), 0.00) AS total_collected
FROM vehicles v
JOIN parking_tickets t ON v.vehicle_id = t.vehicle_id
JOIN payments p ON t.ticket_id = p.ticket_id
WHERE p.payment_status = 'PAID'
GROUP BY v.vehicle_type
ORDER BY total_collected DESC;
3. Relational Subquery (Find Slots with Active Car Parking)
SELECT slot_number, status 
FROM parking_slots 
WHERE current_vehicle_id IN (
    SELECT vehicle_id FROM vehicles WHERE vehicle_type = 'CAR'
);
